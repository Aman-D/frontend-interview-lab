// Globals provided by the Frontend Interview Lab sandbox test harness (public/runner/harness.js).
interface FilMock<A extends any[] = any[], R = any> {
  (...args: A): R;
  mock: { calls: A[]; results: { type: 'return' | 'throw'; value: unknown }[]; contexts: unknown[] };
  mockImplementation(f: (...args: A) => R): FilMock<A, R>;
  mockImplementationOnce(f: (...args: A) => R): FilMock<A, R>;
  mockReturnValue(v: R): FilMock<A, R>;
  mockResolvedValue(v: unknown): FilMock<A, R>;
  mockRejectedValue(v: unknown): FilMock<A, R>;
  mockClear(): FilMock<A, R>;
}
interface FilMatchers<R = void> {
  toBe(v: unknown): R; toEqual(v: unknown): R; toStrictEqual(v: unknown): R;
  toBeTruthy(): R; toBeFalsy(): R; toBeNull(): R; toBeUndefined(): R; toBeDefined(): R;
  toBeInstanceOf(c: Function): R; toBeGreaterThan(n: number): R; toBeGreaterThanOrEqual(n: number): R;
  toBeLessThan(n: number): R; toBeLessThanOrEqual(n: number): R; toContain(v: unknown): R; toHaveLength(n: number): R;
  toMatch(re: RegExp | string): R; toThrow(e?: string | RegExp | Function): R;
  toHaveBeenCalled(): R; toHaveBeenCalledTimes(n: number): R; toHaveBeenCalledWith(...args: unknown[]): R; toHaveBeenLastCalledWith(...args: unknown[]): R;
  toBeInTheDocument(): R; toBeVisible(): R; toHaveTextContent(t: string | RegExp): R; toHaveAttribute(name: string, value?: unknown): R;
  toHaveFocus(): R; toBeDisabled(): R; toHaveClass(c: string): R; toHaveValue(v: string): R;
}
